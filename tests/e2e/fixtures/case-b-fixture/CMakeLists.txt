cmake_minimum_required(VERSION 3.16)
project(ircount CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ircount STATIC src/ircount.cpp)
target_include_directories(ircount PUBLIC include)
add_executable(test_ircount tests/test_ircount.cpp)
target_link_libraries(test_ircount PRIVATE ircount)
enable_testing()
add_test(NAME ircount COMMAND test_ircount)
